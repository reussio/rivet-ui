import { defineComponentMetadata } from "@/content/components/schema"

export const metadata = defineComponentMetadata({
  "name": "Search Field",
  "slug": "search-field",
  "description": "Collects a search query with native form behavior, clear controls, loading feedback, and controlled or uncontrolled state."
})

<ComponentUsage>

```tsx
import { SearchField } from "@/components/ui/search-field"
import { Field, FieldLabel } from "@/components/ui/field"

<Field>
  <FieldLabel htmlFor="docs-search">Search documentation</FieldLabel>
  <SearchField
    id="docs-search"
    name="query"
    placeholder="Search components…"
  />
</Field>
```

Use `onValueChange` when controlling the value. The clear button and <kbd>Escape</kbd> both emit an empty string through the same callback.

Set `loading` while results are being resolved. The indicator is non-blocking, so the query remains editable and clearable.

</ComponentUsage>
