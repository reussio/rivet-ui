<ComponentUsage>

```tsx
import { Combobox, ComboboxContent, ComboboxInput, ComboboxInputGroup, ComboboxItem, ComboboxLabel, ComboboxList, ComboboxTrigger } from "@/components/ui/combobox"

<Combobox items={regions} autoHighlight>
  <ComboboxLabel htmlFor="deployment-region">Deployment region</ComboboxLabel>
  <ComboboxInputGroup>
    <ComboboxInput id="deployment-region" placeholder="Search regions" />
    <ComboboxTrigger aria-label="Open regions" />
  </ComboboxInputGroup>
  <ComboboxContent>
    <ComboboxList>
      {(region) => (
        <ComboboxItem key={region} value={region}>{region}</ComboboxItem>
      )}
    </ComboboxList>
  </ComboboxContent>
</Combobox>
```
