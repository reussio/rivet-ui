import { defineComponentMetadata } from "@/content/components/schema"

export const metadata = defineComponentMetadata({
  "name": "Keyboard Shortcut",
  "slug": "keyboard-shortcut",
  "description": "Displays platform-aware key combinations and registers matching keyboard listeners.",
  "status": "beta",
  "since": "0.1.0"
})

<ComponentUsage>

```tsx
"use client"

import { KeyboardShortcut, useKeyboardShortcut } from "@/components/ui/keyboard-shortcut"

function CommandMenuShortcut() {
  useKeyboardShortcut(["mod", "K"], openCommandMenu)

  return <KeyboardShortcut keys={["mod", "K"]} />
}
```

</ComponentUsage>

### KeyboardShortcutScript

Render `KeyboardShortcutScript` once in the root layout's `<head>` to resolve automatic platform labels before the first paint.

| Prop | Type | Default |
