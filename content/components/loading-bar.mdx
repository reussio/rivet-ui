import { defineComponentMetadata } from "@/content/components/schema"

export const metadata = defineComponentMetadata({
  "name": "Loading Bar",
  "slug": "loading-bar",
  "description": "Shows global navigation progress and exposes application-wide loading controls.",
  "status": "stable",
  "since": "0.1.0"
})

<ComponentUsage>

```tsx
"use client"

import { LoadingBarProvider, useLoadingBar } from "@/components/ui/loading-bar"

function Navigation() {
  const loadingBar = useLoadingBar()

  return <button onClick={() => loadingBar.start()}>Navigate</button>
}

<LoadingBarProvider>
  <Navigation />
</LoadingBarProvider>
```

</ComponentUsage>
